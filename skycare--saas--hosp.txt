CREATE TABLE tenants (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name TEXT NOT NULL,
    slug TEXT UNIQUE NOT NULL,
    email TEXT,
    phone TEXT,
    address TEXT,
    logo_url TEXT,
    plan TEXT DEFAULT 'basic',
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()

🔹 3. USERS & ROLES
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
    email TEXT UNIQUE NOT NULL,
    password_hash TEXT,
    full_name TEXT,
    role TEXT CHECK (role IN ('super_admin','hospital_admin','doctor','nurse','pharmacist','lab_tech','cashier')),
    phone TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_users_tenant ON users(tenant_id);
🔹 4. PATIENTS
CREATE TABLE patients (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,

    gender TEXT,
    dob DATE,
    phone TEXT,
    email TEXT,
    address TEXT,
    blood_group TEXT,
    genotype TEXT,
    allergies TEXT,
    nhia_number TEXT,
    insurance_provider TEXT,
    next_of_kin TEXT,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_patients_tenant ON patients(tenant_id);
CREATE INDEX idx_patients_hospital_id ON patients(hospital_patient_id);
🔹 5. APPOINTMENTS
CREATE TABLE appointments (

    notes TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_appointments_tenant ON appointments(tenant_id);

🧠 HELPER FUNCTIONS (VERY IMPORTANT)

Create reusable functions to simplify policies 👇

🔹 Function: Get Tenant ID
CREATE OR REPLACE FUNCTION get_tenant_id()
RETURNS UUID AS $$
SELECT (auth.jwt() ->> 'tenant_id')::uuid;
$$ LANGUAGE sql STABLE;
🔹 Function: Get Role
CREATE OR REPLACE FUNCTION get_user_role()
RETURNS TEXT AS $$
SELECT auth.jwt() ->> 'role';
$$ LANGUAGE sql STABLE;
🔹 Function: Is Admin
CREATE OR REPLACE FUNCTION is_admin()
RETURNS BOOLEAN AS $$
SELECT get_user_role() IN ('super_admin', 'hospital_admin');
$$ LANGUAGE sql STABLE;

💰 Revenue Trigger
CREATE OR REPLACE FUNCTION update_daily_revenue()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO analytics_daily (tenant_id, branch_id, date, total_revenue)
  VALUES (NEW.tenant_id, NEW.branch_id, CURRENT_DATE, NEW.amount)
  ON CONFLICT (tenant_id, branch_id, date)
  DO UPDATE SET total_revenue = analytics_daily.total_revenue + NEW.amount;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_revenue
AFTER INSERT ON payments
FOR EACH ROW
EXECUTE FUNCTION update_daily_revenue();
📅 Appointment Trigger
CREATE OR REPLACE FUNCTION update_appointments()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO analytics_daily (tenant_id, branch_id, date, total_appointments)
  VALUES (NEW.tenant_id, NEW.branch_id, CURRENT_DATE, 1)
  ON CONFLICT (tenant_id, branch_id, date)
  DO UPDATE SET total_appointments = analytics_daily.total_appointments + 1;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_appointments
AFTER INSERT ON appointments
FOR EACH ROW
EXECUTE FUNCTION update_appointments();

// components/calendar/AppointmentCalendar.tsx
'use client'

import FullCalendar from '@fullcalendar/react'
import timeGridPlugin from '@fullcalendar/timegrid'
import interactionPlugin from '@fullcalendar/interaction'
import dayGridPlugin from '@fullcalendar/daygrid'
import { useEffect, useState } from 'react'
import { supabase } from '@/lib/supabaseClient'
import AppointmentModal from './AppointmentModal'

export default function AppointmentCalendar() {
  const [events, setEvents] = useState<any[]>([])
  const [selected, setSelected] = useState<any>(null)

  useEffect(() => {
    fetchAppointments()
    subscribeToChanges()
  }, [])

  const fetchAppointments = async () => {
    const { data } = await supabase
      .from('appointments')
      .select('*')

    const formatted = data?.map((a) => ({
      id: a.id,
      title: a.patient_name,
      start: a.start_time,
      end: a.end_time,
    }))

    setEvents(formatted || [])
  }

  const subscribeToChanges = () => {
    supabase
      .channel('appointments')
      .on(
        'postgres_changes',
        { event: '*', schema: 'public', table: 'appointments' },
        fetchAppointments
      )
      .subscribe()
  }

  const handleDateSelect = (info: any) => {
    setSelected({
      start: info.startStr,
      end: info.endStr,
    })
  }

  const handleEventDrop = async (info: any) => {
    await supabase
      .from('appointments')
      .update({
        start_time: info.event.start,
        end_time: info.event.end,
      })
      .eq('id', info.event.id)
  }

  return (
    <div className="bg-white p-4 rounded shadow">
      <FullCalendar
        plugins={[timeGridPlugin, interactionPlugin, dayGridPlugin]}
        initialView="timeGridWeek"
        selectable
        editable
        events={events}
        select={handleDateSelect}
        eventDrop={handleEventDrop}
        eventResize={handleEventDrop}
        height="80vh"
      />

      {selected && (
        <AppointmentModal
          data={selected}
          onClose={() => setSelected(null)}
          onSave={fetchAppointments}
        />
      )}
    </div>
  )
}

  }

  return (
    <div className="fixed inset-0 bg-black/30 flex items-center justify-center">
      <div className="bg-white p-6 rounded shadow w-96">
        <h2 className="text-lg font-bold mb-4">New Appointment</h2>

        <input
          type="text"
          placeholder="Patient Name"
          className="w-full border p-2 mb-3"
          value={patientName}
          onChange={(e) => setPatientName(e.target.value)}
        />

        <div className="flex justify-end gap-2">
          <button onClick={onClose}>Cancel</button>
          <button
            onClick={handleSave}
            className="bg-blue-500 text-white px-3 py-1 rounded"
          >
            Save
          </button>
        </div>
      </div>
    </div>
  )
}
